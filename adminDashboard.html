Firestore Security Rules

// Firestore Production Security Rules rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { match /loanApplications/{applicationId} { allow create: if request.auth == null; allow read, update, delete: if false; // Additional validation for data structure allow create: if request.resource.data.brand is string && request.resource.data.model is string && request.resource.data.color is string && request.resource.data.yearsToPay is string && request.resource.data.downpayment is number && request.resource.data.firstName is string && request.resource.data.lastName is string && request.resource.data.birthdate is string && request.resource.data.age is number; } match /adminUsers/{userId} { allow read: if request.auth != null; allow write: if false; } } }

Storage Security Rules

// Storage Production Security Rules rules_version = '2'; service firebase.storage { match /b/{bucket}/o { match /id_documents/{fileName} { allow create: if request.auth == null && request.resource.size < 5 * 1024 * 1024 && // 5MB request.resource.contentType.matches('image/.*|application/pdf'); allow read, update, delete: if false; } } }
These rules allow public form submissions but restrict access to:
  • Prevent reading of application data
  • Block modification/deletion of records
  • Validate data structure on submission
  • Limit file uploads to 5MB image/PDF files
